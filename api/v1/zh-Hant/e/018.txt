// 形狀轉場 / Shape transition
// 包: motion (Motion Graphics 動態手法) · element 18 · 分類: 轉場與吸睛效果
// 會發生什麼: 彩色色帶掃過並覆蓋畫面，離開後已換成下一個場景。
// 動態規格:
//   - 白天 → 夜晚 → 白天，各 2.35 秒。4 條橫向色帶（高 52；藍色、珊瑚色、黃色、薄荷綠），兩端斜切 80 px
//   - 覆蓋：0.38–1.0 秒。每條色帶的前緣從畫面一側移到另一側。色帶之間依序延遲該階段的 14%，每條以 easeInOut（3 次方）走完該階段的 58%
//   - 1.0 秒（全部覆蓋的瞬間）將底下的場景換成下一個
//   - 露出：1.12–1.75 秒。每條色帶的後緣以相同延遲與曲線跟上，露出下一個場景
//   - 方向：白天→夜晚為由左到右，夜晚→白天則左右反轉
// 使用時機: 想用品牌色的圖形切換場景時使用，例如章節分隔、話題轉換。
// 標籤: shape-transition, transition, wipe, bands, stripes, cover, scene-change
// 尺寸: 539x208 px · 一輪: 4.7 s
// Needs: kit motion -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/motion.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#018
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 18
// 18 シェイプトランジション（色のついた帯が画面を覆い隠し、抜けると次の画面になっている）
CELLS[17] = abCell((k, w, h, u, a, b) => {
  const p1 = P(u, TA - 0.12, TA + 0.5), p2 = P(u, TA + 0.62, TA + 1.25), N = 4, sk = 80;
  SCR[p1 >= 1 ? b : a](k, w, h);
  const cols = [C.blue, C.coral, C.yel, C.mint];
  for (let i = 0; i < N; i++) {
    const d = i * 0.14, e1 = eIO(clamp((p1 - d) / (1 - 0.42))), e2 = eIO(clamp((p2 - d) / (1 - 0.42)));
    if (e1 <= 0 || e2 >= 1) continue;
    const bh = h / N + 1, y = i * h / N, span = w + sk + 20, dir = a === 0 ? 1 : -1;
    let xa = -sk - 10 + e2 * span, xb = -sk - 10 + e1 * span;
    k.save(); if (dir < 0) { k.translate(w, 0); k.scale(-1, 1); }
    k.fillStyle = cols[i]; k.beginPath(); k.moveTo(xa + sk, y); k.lineTo(xb + sk, y); k.lineTo(xb, y + bh); k.lineTo(xa, y + bh); k.closePath(); k.fill(); k.restore();
  }
});
