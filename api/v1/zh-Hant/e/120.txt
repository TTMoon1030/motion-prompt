// 多重曝光 / Double exposure
// 包: photo (攝影風格：光線、鏡頭與調色) · element 24 · 分類: 調色與風格
// 會發生什麼: 背景變成白色，縮小後的山丘、房子、樹與太陽的風景只出現在胸像的形狀裡。
// 動態規格:
//   - 場景 3：左側為胸像（頭部中心 x 140, y 82，半徑 28 px），右側為山丘、房子、樹與太陽（x 432, y 54，半徑 15 px）。
//   - 背景：胸像以外為白色 #FFFFFF
//   - 胸像內：同一幅風景縮為 0.7×，擺放成太陽位在頭部右上、山丘稜線橫過脖子一帶
//   - 內部顏色：天空 #D3DBE6，山丘 #9AA8BD 與 #76869F，地面 #55647E，樹 #3F4C66
//   - 切換：在原本的畫面上，依效果量疊上多重曝光的畫面
//   - 時間：效果在 0.4–1.4 秒進入，維持到 4.0 秒，4.8 秒前恢復原狀（一個循環 5 秒，三次方緩入緩出，約等於 cubic-bezier(0.65,0,0.35,1)）
// 使用時機: 用於想在一張圖裡把人物與地點、回憶或內心世界連結起來的海報或封面。
// 標籤: double exposure, multiple exposure, blend, silhouette, surreal, composite
// 尺寸: 539x208 px · 一輪: 5 s
// Needs: kit photo -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/photo.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#120
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 24
// 二重露光: 背景が白く飛び、胸像の形の中にだけ丘と家と木と太陽が写り込む
CELLS[23] = cell((k, u) => {
  const a = env(u);
  const DX = Object.assign({}, BASE, { sky: '#D3DBE6', sky2: '#D3DBE6', far: '#9AA8BD', mid: '#76869F', ground: '#55647E', bld: '#E8ECF2', edge: '#55647E', win: '#9AA8BD', winOff: '#9AA8BD', tree: '#3F4C66', trunk: '#3F4C66' });
  const draw = kk => {
    kk.fillStyle = '#FFFFFF'; kk.fillRect(0, 0, CW, CH);
    kk.save(); bustPath(kk, QX, QY, QR); kk.clip();
    // 風景を0.7倍にして、太陽が頭の右上、丘の稜線が首のあたりに来るように置く
    kk.translate(QX, QY); kk.scale(0.7, 0.7); kk.translate(-415, -77);
    kk.fillStyle = DX.sky; kk.fillRect(0, -200, CW * 2, CH * 3);
    stageLand(kk, DX, { person: false });
    kk.fillStyle = DX.ground; kk.fillRect(-CW, 200, CW * 3, CH * 3);
    kk.restore();
  };
  if (a >= 0.999) { draw(k); return; }   // 効果が入りきったら直接描く（拡大表示でも輪郭がくっきり）
  stageLand(k, BASE);
  if (a < 0.002) return;
  const [L, kl] = layer('dx'); draw(kl);   // 入り・戻りの途中は、別の面に描いて重ねる
  k.save(); k.globalAlpha = a; k.drawImage(L, 0, 0); k.restore();
});
