// 蝴蝶光 / Butterfly lighting
// 包: photo (攝影風格：光線、鏡頭與調色) · element 06 · 分類: 打光方式與時段
// 會發生什麼: 臉部正前方高處的光亮起，臉的左右兩側均勻受光，只在下巴下方落下一小塊左右對稱的陰影。
// 動態規格:
//   - 場景 1：地平線（y 150）前方是胸部以上的人物（頭部中心 x 270, y 78，半徑 29 px，不畫五官與頭髮）。背景轉為暗一階的場景（天空 #E7E5E1）
//   - 頭部：整顆受光 #FFC75A（沒有陰影側）
//   - 下巴下方的陰影：橢圓，中心 y 78+1.06R，寬 0.72R、高 0.38R（R = 29，約 21×11 px），顏色 #33405A
//   - 身體：中心 ±0.8×56 px 以內受光，外側的細帶維持原色；燈位圖中光在相機側（0°），右上顯示「0° · ↑45°」
//   - 時間：效果在 0.4–1.4 秒進入，維持到 4.0 秒，4.8 秒前恢復原狀（一個循環 5 秒，三次方緩入緩出，約等於 cubic-bezier(0.65,0,0.35,1)）
// 使用時機: 用於想把五官拍得好看的美妝、時尚人像。
// 標籤: butterfly lighting, paramount lighting, beauty lighting, portrait lighting, overhead light, glamour
// 尺寸: 539x208 px · 一輪: 5 s
// Needs: kit photo -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/photo.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#102
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 06
// バタフライライティング: 顔の正面・高い位置からの光。顔は左右対称に明るく、あごの下にだけ小さな影
CELLS[5] = cell((k, u) => {
  const a = env(u), R = PR, W = torsoW(R);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a);
  bust(k, PX, PY, R, BASE.person, kk => {
    torso(kk, PX, PY, R, PX - 0.8 * W, PX + 0.8 * W, lit);
    kk.fillStyle = sh; kk.beginPath(); kk.ellipse(PX, PY + R * 1.06, R * 0.72, R * 0.38, 0, 0, TAU); kk.fill();    // あごの下の影（左右対称）
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, R, 0, TAU); kk.fill();
  });
  plan(k, [[0, a]]);
  hud(k, '0° · ↑45°');
});
