// 手風琴 / Accordion
// 包: ui (UI 元件名稱) · element 04 · 分類: 展開與彈出式元件
// 會發生什麼: 點擊三個標題中的第一個，內容在下方展開並把下面的列往下推；再點一次即收合。
// 動態規格:
//   - 4.5 秒循環，卡片 542×191。三個標題列（寬 494、高 34、圓角 8、間距 6），右端有向下箭頭。
//   - 游標在 0.95 秒抵達第一個標題，於 1.0 秒與 3.3 秒點擊。
//   - 展開量 ex：1.0–1.35 秒以 easeInOutCubic 由 0→1，3.3–3.62 秒同樣由 1→0。
//   - 第一列高度由 34 增加到 92（+58×ex），下方各列隨之往下推；兩行內文裁切在列框內，以不透明度 ex 淡入。
//   - 展開期間，箭頭旋轉 180°×ex，邊框變為淺藍色，標題文字變為藍色。
// 使用時機: 用於常見問題或較長的說明，只列出標題，讓需要的人自行展開內容。
// 標籤: accordion, collapsible, expand, collapse, disclosure, faq, details
// 尺寸: 542x191 px · 一輪: 4.5 s
// Needs: kit ui -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/ui.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#052
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 04
// 04 アコーディオン（Accordion）
CELLS[3] = ui((w, h, t) => {
  const ex = eio(seg(t, 1.0, 1.35)) * (1 - eio(seg(t, 3.3, 3.62)));
  let y = 12; const H0 = 34, EX = 58;
  for (let i = 0; i < 3; i++) {
    const extra = i === 0 ? EX * ex : 0;
    strokeRR(24, y, 494, H0 + extra, 8, i === 0 && ex > .05 ? C.sb : C.line2, 1.5);
    bar(42, y + 12, [180, 220, 150][i], i === 0 && ex > .05 ? C.blue : C.line2, 10);
    chevron(494, y + 17, 6, i === 0 ? Math.PI * ex : 0, i === 0 && ex > .05 ? C.blue : '#9a9896');
    if (i === 0 && ex > 0) { X.save(); rr(24, y, 494, H0 + extra, 8); X.clip(); X.globalAlpha = ex; bar(42, y + H0 + 6, 420); bar(42, y + H0 + 24, 320); X.restore(); }
    y += H0 + extra + 6;
  }
  pointer([[.3, w + 30, h + 30], [.95, 300, 31], [1.6, 300, 31], [3.0, 300, 31], [3.55, 300, 31], [4.2, w + 30, h + 30]], [1.0, 3.3], t);
});
