// 分割光 / Split lighting
// 包: photo (攝影風格：光線、鏡頭與調色) · element 04 · 分類: 打光方式與時段
// 會發生什麼: 光從接近正面（20°）轉到正側面（90°），頭部的陰影從月牙形擴大到剛好一半，臉沿鼻樑位置縱向分成亮暗兩半。
// 動態規格:
//   - 場景 1：地平線（y 150）前方是胸部以上的人物（頭部中心 x 270, y 78，半徑 29 px，不畫五官與頭髮）。背景轉為暗一階的場景（天空 #E7E5E1）
//   - 光：從畫面左側打來，角度 20° → 90°（右下為燈位圖，右上顯示「90° · ↑0°」）
//   - 頭部：x = 270 + cos(角度)·√(29² − dy²) 右側為陰影 #33405A，左側為受光 #FFC75A；90° 時明暗交界是筆直的中線
//   - 身體（半寬 56 px）：x = 270 + cos(角度)·56 右側為陰影，左緣一條帶狀區域受光
//   - 時間：效果在 0.4–1.4 秒進入，維持到 4.0 秒，4.8 秒前恢復原狀（一個循環 5 秒，三次方緩入緩出，約等於 cubic-bezier(0.65,0,0.35,1)）
// 使用時機: 想讓人像帶有力量、緊張感或陰鬱氛圍時。
// 標籤: split lighting, side light, portrait lighting, half face, dramatic, chiaroscuro
// 尺寸: 539x208 px · 一輪: 5 s
// Needs: kit photo -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/photo.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#100
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 04
// スプリットライティング: 光が真横（90°）へ回り、顔の右半分が影に沈む
CELLS[3] = cell((k, u) => {
  const a = env(u), az = lerp(0.35, Math.PI / 2, a);
  stagePortrait(k, pal(BASE, DIM, a));
  const lit = mixc(BASE.person, BASE.lit, a), sh = mixc(BASE.person, BASE.shade, a), W = torsoW(PR);
  bust(k, PX, PY, PR, BASE.person, kk => {
    kk.fillStyle = lit; kk.beginPath(); kk.arc(PX, PY, PR, 0, TAU); kk.fill();
    headShade(kk, PX, PY, PR, az, 1, sh);
    torso(kk, PX, PY, PR, PX - W, PX - Math.cos(az) * 0.4 * W - 0.4 * W, lit);
    torso(kk, PX, PY, PR, PX + Math.cos(az) * W, PX + W, sh);
  });
  plan(k, [[az, a]]);
  hud(k, `${Math.round(az * 180 / Math.PI)}° · ↑0°`);
});
