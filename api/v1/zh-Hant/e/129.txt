// Text Scramble / Text scramble
// 包: motion2 (Motion Graphics 動態手法 第二彈) · element 09 · 分類: 文字與圖形效果
// 會發生什麼: 隨機字元不斷變換，再由左到右依序定成正確的單字。
// 動態規格:
//   - 前 2.35 秒顯示「HELLO」，後半顯示「WORLD」。5 個等寬字型的字母（92 px）。
//   - 尚未定住的格位顯示隨機字元，取自 A–Z（不含 I 與 O）、0–9 和 #$%&@?，每 4.7 秒變換 76 次（約每 0.062 秒），薄荷綠，不透明度 75%。
//   - 由左到右依序定住：第 i 格在 0.45 + 0.18i 秒（最後一格在 1.17 秒）。剛定住的字母先呈黃色 0.22 秒，之後變白色，縮放以 1 + 0.2·e^(−14t) 收斂。
//   - 從 1.85 秒起，由右到左依序變回隨機字元（第 i 格在 1.85 + 0.08 × (4 − i) 秒）。
//   - 每格下方有橫條：已定住為藍色，下一個要定住的為黃色，其餘為深藍色。
// 使用時機: 用來呈現標題、數字或代號，帶出解碼、駭客或科技感。
// 標籤: text-scramble, scramble, decode, shuffle, random, text, hacker
// 尺寸: 539x208 px · 一輪: 4.7 s
// Needs: kit motion2 -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/motion2.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#129
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 09
// 09 テキストスクランブル（ランダムな文字が入れ替わり続け、左から順に正しい単語に決まる）
const SCR_GL = 'ABCDEFGHJKLMNPQRSTUVWXYZ0123456789#$%&@?';
CELLS[8] = (k, w, h, u) => {
  u = snapU(u);
  stage(k);
  const T = CYC / 2, n = Math.floor(u / T), lu = u - n * T, word = n ? 'WORLD' : 'HELLO', N = 5;
  const f = `700 92px Menlo, "SF Mono", monospace`; k.font = f;
  const cw = k.measureText('M').width + 16, x0 = w / 2 - cw * (N - 1) / 2, Y = h / 2 - 8;
  const step = Math.floor(u / (CYC / 76));
  for (let i = 0; i < N; i++) {
    const tl = 0.45 + i * 0.18, tu = 1.85 + (N - 1 - i) * 0.08, locked = lu >= tl && lu < tu;
    let ch, col, sc = 1;
    if (locked) { ch = word[i]; const age = lu - tl; col = age < 0.22 ? C.yel : '#fff'; sc = 1 + 0.2 * Math.exp(-age * 14); }
    else { ch = SCR_GL[Math.floor(H01(step * 131 + i * 977 + n * 31) * SCR_GL.length)]; col = C.mint; }
    k.save(); k.translate(x0 + i * cw, Y); k.scale(sc, sc); k.globalAlpha = locked ? 1 : 0.75; txt(k, ch, 0, 0, f, col, 'center', 'middle'); k.restore();
    const next = !locked && lu < tl && (i === 0 || lu >= 0.45 + (i - 1) * 0.18);
    k.fillStyle = locked ? C.blue : next ? C.yel : C.navy3; rr(k, x0 + i * cw - cw / 2 + 8, Y + 58, cw - 16, 8, 4); k.fill();
  }
};
