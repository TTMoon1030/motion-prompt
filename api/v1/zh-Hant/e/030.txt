// 環繞運鏡 / Orbit (Arc shot)
// 包: camera (運鏡與轉場) · element 06 · 分類: 鏡頭運動
// 會發生什麼: 攝影機沿著以被攝體為圓心的圓周移動；人物保持在中央，後方的樹和建築大幅橫向掠過。
// 動態規格:
//   - 攝影機：在以人物（正前方 9 m）為圓心、半徑 6 m 的圓周上移動，-0.95 rad → +0.95 rad（-54° → +54°，共約 109°）
//   - 方向：始終朝向圓心（偏航角 = -角度）；高度 1.7 m，俯仰角固定為 -0.07 rad（約向下 4°）
//   - 時間：0.5 秒到 3.6 秒（3.1 秒），三次緩入緩出（相當於 cubic-bezier(0.65,0,0.35,1)）
//   - 被攝體：人物不動；臉上的白點隨角度橫向偏移，看得出攝影機在繞行；腳下地面有半徑 1.3 m 的虛線圓
//   - 背景：12 棵樹等距排在半徑 10 m 的圓上，攝影機繞行時從人物後方橫越
//   - 復位：3.6–4.2 秒停住，最後 0.5 秒交叉淡化回起始狀態；右上角顯示角度（-54° → +54°）
// 使用時機: 要立體呈現人物或商品，或強調登場、關鍵時刻時。
// 標籤: orbit, arc shot, circle, 360, rotate around, revolve, camera move
// 尺寸: 539x208 px · 一輪: 9.4 s
// Needs: kit camera -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/camera.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#030
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 06
// 06 オービット: 被写体を中心に円を描いて回り込む
CELLS[5] = shot(camCell((k, w, h, p) => {
  const a = lerp(-0.95, 0.95, p), R = 6, P0 = [0, 0, 9];
  const c = cam({ x: P0[0] + Math.sin(a) * R, z: P0[2] - Math.cos(a) * R, yaw: -a, y: 1.7, pitch: -0.07 });
  world(k, w, h, c, {
    person: { z: 9, face: a }, ring: [0, 9, 10], buildings: true, trees: false,
    after: (kk, cc) => { kk.strokeStyle = C.blueM; kk.lineWidth = 2; kk.setLineDash([5, 5]); kk.beginPath(); for (let i = 0; i <= 48; i++) { const b = i / 48 * Math.PI * 2; const q = proj(cc, Math.sin(b) * 1.3, 0, 9 + Math.cos(b) * 1.3); i ? kk.lineTo(q.x, q.y) : kk.moveTo(q.x, q.y); } kk.stroke(); kk.setLineDash([]); }
  });
}, p => `${Math.round(lerp(-0.95, 0.95, p) * 57.3)}°`.replace(/^(\d)/, '+$1'), 0.5, 3.6));
