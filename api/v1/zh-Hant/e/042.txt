// 匹配剪接 / Match cut
// 包: camera (運鏡與轉場) · element 18 · 分類: 轉場
// 會發生什麼: 拋起的球在下一個鏡頭變成同樣位置、同樣大小的太陽；兩個不同的場景以相同的形狀銜接。
// 動態規格:
//   - 銜接：在 2.3 秒直接切換（0 影格）；切換前後藍色圓形的位置（中心往右 120 px、往上 46 px）與半徑 17 px 一致
//   - 前一個鏡頭（0–2.3 秒）：固定攝影機（高度 1.4 m）；人物（左 1.2 m、距離 7 m）從 0.5 秒開始舉起雙臂把球拋出，三次緩出（相當於 cubic-bezier(0.33,1,0.68,1)）
//   - 球：從人物頭頂上方（高 2.3 m）移動到對位點；半徑 5 px → 17 px，路徑是最多往上鼓起 30 px 的弧線
//   - 後一個鏡頭（2.3–4.7 秒）：另一個地點（沒有建築，高度 1.2 m）；同一位置的藍色圓形變成太陽，在 2.4 秒內下沉 14 px
//   - 標示：半徑 26 px 的藍色虛線圓，只在切點前後 0.7 秒內淡入淡出顯示
//   - 右上角顯示「SHOT 1」→「SHOT 2」；在 4.7 秒直接切回 SHOT 1
// 使用時機: 要讓時間或地點相隔很遠的兩個場景，藉由共同的形狀或動作連結起來時。
// 標籤: match cut, graphic match, shape match, cut, transition, edit
// 尺寸: 539x208 px · 一輪: 9.4 s
// Needs: kit camera -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/camera.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#042
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 18
// 18 マッチカット（投げたボール → 同じ位置・大きさの太陽）
CELLS[17] = shot((k, w, h, u) => {
  const cut = 2.3;
  const BX = 120, BY = -46, BR = 17;
  if (u < cut) {
    const q = eOut(P(u, 0.5, cut));
    world(k, w, h, cam({ y: 1.4 }), {
      person: { x: -1.2, z: 7, armUp: q > 0.05 }, sun: null,
      after: (kk, cc) => { const s = proj(cc, -1.2, 2.3, 7); const x = lerp(s.x, BX, q), y = lerp(s.y, BY, q) - Math.sin(q * Math.PI) * 30, r = lerp(5, BR, q); kk.fillStyle = C.blue; kk.beginPath(); kk.arc(x, y, r, 0, 7); kk.fill(); }
    });
  } else {
    const q = P(u, cut, CYC);
    world(k, w, h, cam({ y: 1.2 }), {
      buildings: false, sun: null, trees: true,
      after: (kk) => { kk.fillStyle = C.blue; kk.beginPath(); kk.arc(BX, BY + q * 14, BR, 0, 7); kk.fill(); }
    });
  }
  // 形がそろっていることを示す点線の円
  const g = Math.max(0, 1 - Math.abs(u - cut) / 0.7);
  if (g > 0) { k.save(); k.translate(w / 2, h / 2); k.globalAlpha = g; k.strokeStyle = C.blue; k.lineWidth = 2; k.setLineDash([5, 4]); k.beginPath(); k.arc(BX, BY + (u > cut ? P(u, cut, CYC) * 14 : 0), BR + 9, 0, 7); k.stroke(); k.restore(); }
  return u < cut ? 'SHOT 1' : 'SHOT 2';
});
