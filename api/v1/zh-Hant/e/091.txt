// 無限捲動 / Infinite scroll
// 包: lp (Landing Page 與網站效果) · element 19 · 分類: 載入、轉場與持續動態
// 會發生什麼: 清單往下捲到底時出現旋轉的載入圖示，接著新增三筆項目，可以繼續往下讀。
// 動態規格:
//   - 項目清單（331 × 44，圓角 8，間隔 52），最後一個項目下方有載入圖示（半徑 10，每 0.8 秒轉一圈）。
//   - 捲動量：0.25–0.85 秒 38→116 px，1.15–1.75 秒 116→194 px（三次方 easeInOut）；停在看得到清單末端的位置。
//   - 2.6 秒新增三筆：每筆在 0.35 秒內淡入（不透明度 0→1），並從下方 10 px 往上移（三次方 easeOut）。載入圖示移到新的末端。
//   - 每個週期前進三筆（156 px）。內容每三筆重複一次，所以週期結束時的畫面與開始時相同。
// 使用時機: 用在社群動態或最新消息列表等，希望使用者不必換頁就一直讀下去的地方。
// 標籤: infinite-scroll, endless-scroll, load-more, feed, pagination, spinner, list
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#091
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 19
// 19 無限スクロール（いちばん下まで来ると読み込み表示が出て、続きが足される）
CELLS[18] = (k, w, h, u) => {
  frame(k);
  // 1周期で 3件（156px）ぶん進む。周期の終わりは始まりと同じ見た目（内容は3件ごとにくり返す）
  const PITCH = 52, BATCH = 3, L0 = 6, TL = 2.6;
  const y = steps([[0.25, 38], [0.85, 116], [1.15, 116], [1.75, 194], [4.8, 194]], u);
  const n = u < TL ? L0 : L0 + BATCH, end = 10 + n * PITCH;
  view(k, () => {
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH);
    k.save(); k.translate(0, -y);
    for (let i = 0; i < n; i++) {
      const top = 10 + i * PITCH, v = i % 3, fresh = i >= L0 ? eOut(P(u, TL, TL + 0.35)) : 1;
      if (top - y > VH || top + 44 - y < 0) continue;
      k.save(); k.globalAlpha = fresh; k.translate(0, 10 * (1 - fresh));
      k.fillStyle = '#fff'; rr(k, 90, top, 331, 44, 8); k.fill(); pic(k, 98, top + 6, 48, 32, v, 5);
      bars(k, 158, top + 12, [180, 140, 200][v], 1, C.ink, 7); bars(k, 158, top + 26, [120, 150, 100][v], 1, C.faint, 6);
      k.restore();
    }
    spinner(k, VW / 2, end + 18, 10, u);
    k.restore();
  });
};
