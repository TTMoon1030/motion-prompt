// 逆光 / Backlighting
// 包: photo (攝影風格：光線、鏡頭與調色) · element 01 · 分類: 打光方式與時段
// 會發生什麼: 太陽從畫面外繞到人物頭部正後方，人物輪廓亮起一圈細細的光邊，整個畫面蒙上一層淡淡的暖色光，面向相機的一側沉入暗部。
// 動態規格:
//   - 場景 1：地平線（y 150）前方是胸部以上的人物（頭部中心 x 270, y 78，半徑 29 px，不畫五官與頭髮）。受光面為琥珀色 #FFC75A 的平塗色塊，陰影為更深一階的 #33405A 平塗色塊
//   - 太陽：半徑 20 px，從左上的畫面外（x 40, y −40）移到頭部正後方（x 270, y 56），周圍有半徑 2.6× 與 1.7× 的平塗光環（#FFE3A3，35% 與 60%）
//   - 輪廓：先用 7 px 寬的線描出胸像外形再填色，只留下外側 3.5 px 的亮邊
//   - 人物 #4D5A75 → #33405A；天空 #FAF9F7 → #FFF4DC；整個畫面疊上 22% 的 #FFE3A3
//   - 時間：效果在 0.4–1.4 秒進入，維持到 4.0 秒，4.8 秒前恢復原狀（一個循環 5 秒，三次方緩入緩出，約等於 cubic-bezier(0.65,0,0.35,1)）
// 使用時機: 想拍出明亮柔和、被光包圍的人像或畫面時。
// 標籤: backlighting, backlit, contre-jour, sun behind, haze, glow, halo
// 尺寸: 539x208 px · 一輪: 5 s
// Needs: kit photo -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/photo.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#097
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 01
// 逆光: 太陽が画面の外から頭の真後ろへ回り込み、輪郭が光って人物は暗くなる
CELLS[0] = cell((k, u) => {
  const a = env(u);
  const BACK = Object.assign({}, BASE, { sky: '#FFF4DC', sky2: '#FFEBC4', far: '#F6E2BE', ground: '#F7EAD3', bld: '#F7E9D2', winOff: '#EEDCBF', win: '#EEDCBF', tree: '#EEDCBF', edge: '#E3CFAE', post: '#CDB894', trunk: '#E3CFAE', bulb: '#E3CFAE' });
  stagePortrait(k, pal(BASE, BACK, a), { sun: [lerp(40, PX, a), lerp(-40, PY - 22, a), 20, 1] });
  // 輪郭の光: 胸像の形を太い線でなぞってから中を塗り、縁だけを残す
  k.save(); k.globalAlpha = a; bustPath(k, PX, PY, PR); k.strokeStyle = BASE.lit; k.lineWidth = 7; k.lineJoin = 'round'; k.stroke(); k.restore();
  bust(k, PX, PY, PR, mixc(BASE.person, BASE.shade, a));
  // かすみ: 全体に光の色を薄く重ねる
  k.save(); k.globalAlpha = 0.22 * a; k.fillStyle = BASE.glow; k.fillRect(0, 0, CW, CH); k.restore();
});
