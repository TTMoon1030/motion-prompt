// 包: ui (UI 元件名稱) · element 03 · 分類: 展開與彈出式元件
// 會發生什麼: 手指點按「查看詳情」後，面板從底部往上升起；把上緣的拖曳把手往下拉即關閉。
// 動態規格:
//   - 4.5 秒循環，卡片 542×191。手指（半徑 17 的半透明圓形，白色外框）在 1.0 秒點按下方中央的「查看詳情」藍色膠囊形按鈕（170×36）。
//   - 開啟量 a：1.05–1.45 秒以 easeOutCubic 由 0→1。整張卡片鋪上 rgba(18,20,28) 遮罩，不透明度 0.36×a。
//   - 面板：白色、高 136、上方兩角圓角 18，上緣 y = 191 − 136×a。陰影 blur 22、y 偏移 −4（向上）。上緣中央有 52×6 的拖曳把手。
//   - 面板內：一條粗的標題線，以及兩列附 26×26 圖示的項目。
//   - 關閉：手指在 2.75 秒按住把手 (271, 68)，3.0–3.6 秒往下拖；a 乘上 (1 − easeInOutCubic(3.0–3.6 秒))，面板跟著手指從底部消失。
// 使用時機: 在手機上想保留目前畫面、同時顯示詳細內容或選項時使用。
// 標籤: bottom-sheet, sheet, modal-sheet, mobile, slide-up, drag-to-dismiss, panel
// 尺寸: 542x191 px · 一輪: 4.5 s
// Needs: kit ui -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/ui.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#051
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 03
// 03 ボトムシート（Bottom sheet）
CELLS[2] = ui((w, h, t) => {
  pageLines(24, 16, 494); fillRR(24, 76, 239, 54, 8, C.block); fillRR(279, 76, 239, 54, 8, C.block);
  fillRR(186, 144, 170, 36, 18, C.blue); text('詳細を見る', 271, 162, 15, '#fff', 600, 'center');
  const open = eo(seg(t, 1.05, 1.45));
  const drag = eio(seg(t, 3.0, 3.6));
  const a = open * (1 - drag);
  if (open > 0 && a > 0.001) {
    X.fillStyle = `rgba(18,20,28,${.36 * a})`; X.fillRect(0, 0, w, h);
    const sh = 136, sy = h - sh * a;
    shadowed(() => { rr(0, sy, w, sh + 20, 18); X.fillStyle = '#fff'; X.fill(); }, 22, -4, .2);
    fillRR(w / 2 - 26, sy + 10, 52, 6, 3, C.line2);
    bar(28, sy + 30, 160, C.ink, 11);
    for (let i = 0; i < 2; i++) { const y = sy + 56 + i * 36; fillRR(28, y, 26, 26, 6, i ? C.mb : C.blue); bar(66, y + 4, 220, C.line); bar(66, y + 17, 140, C.line, 7); }
  }
  const sy = h - 136 * a;
  finger([[.3, 271, 162], [1.6, 271, 162], [2.75, w / 2, h - 136 + 13], [3.0, w / 2, h - 136 + 13], [3.6, w / 2, h + 10]], [1.0, 3.0], t, (t > .6 && t < 1.6) || (t > 2.75 && t < 3.6));
});
