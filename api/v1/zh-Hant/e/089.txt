// Loading 畫面 / Preloader
// 包: lp (Landing Page 與網站效果) · element 17 · 分類: 載入、轉場與持續動態
// 會發生什麼: 深藍色畫面上，Logo 下方的數字從 0% 跑到 100%，接著畫面往上移開，下方的頁面淡入出現。
// 動態規格:
//   - 填滿視窗的深藍色畫面，上面有 Logo、數字（28 px）與進度條（180 × 4）。
//   - 0.25–2.1 秒 0→100%（三次方 easeOut，取整數）；進度條以相同比例填滿橘紅色。
//   - 2.3–2.85 秒整個畫面往上移 168 px 離開（三次方 easeInOut）。
//   - 頁面的標題與照片在 2.35–2.95 秒淡入（不透明度 0→1），並從下方 14 px 往上移（三次方 easeOut）。
//   - 4.25–4.75 秒畫面從上方降下，回到 0% 的狀態（下一次載入）。
// 使用時機: 在照片或影片很多的網站上，想把載入的等待時間變成展現品牌的時刻時。
// 標籤: preloader, loading-screen, loader, percentage, intro, splash, curtain
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#089
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 17
// 17 ローディング画面（数字が100%まで進むと、幕が上がってページが現れる）
CELLS[16] = (k, w, h, u) => {
  frame(k);
  const n = Math.round(100 * eOut(P(u, 0.25, 2.1)));
  const cy = u < 4.25 ? -VH * eIO(P(u, 2.3, 2.85)) : -VH * (1 - eIO(P(u, 4.25, 4.75)));
  const show = eOut(P(u, 2.35, 2.95));
  view(k, () => {
    header(k, 0);
    k.save(); k.globalAlpha = show; k.translate(0, 14 * (1 - show));
    txt(k, 'ようこそ。', 34, 82, `800 26px ${JP}`, C.ink); bars(k, 34, 98, 200, 3); btn(k, 34, 136, 96, 24, '始める');
    pic(k, 270, 46, 214, 112, 0, 8); k.restore();
    if (cy > -VH) {
      k.save(); k.translate(0, cy); k.fillStyle = C.ink; k.fillRect(0, 0, VW, VH);
      logo(k, VW / 2 - 40, 56, '#fff');
      txt(k, (u >= 4.25 ? 0 : n) + '%', VW / 2, 98, `800 28px ${EN}`, '#fff', 'center', 'middle');
      k.fillStyle = 'rgba(255,255,255,.15)'; rr(k, VW / 2 - 90, 122, 180, 4, 2); k.fill();
      k.fillStyle = C.acc; rr(k, VW / 2 - 90, 122, 180 * (u >= 4.25 ? 0 : n / 100), 4, 2); k.fill();
      k.restore();
    }
  });
};
