// 橫向捲動 / Horizontal scroll
// 包: lp (Landing Page 與網站效果) · element 05 · 分類: 捲動時的效果
// 會發生什麼: 頁面往下捲動時，區塊保持不動，四張附編號的作品卡片往左移動。
// 動態規格:
//   - 捲動量：0.3–3.0 秒 0→420 px，3.5–4.5 秒 420→0（三次方 easeInOut）。
//   - 標題「WORKS」往上捲走；區塊在捲動量 40 px 時抵達頂端並停住。
//   - 卡片：四張 200 × 112，間距 20，左右留白 34。橫向移動量 = 34·2 + 200·4 + 20·3 − 511 = 417 px。
//   - 卡片 x = 起始 x − p × 417，p = (捲動量 − 40) / 380：與捲動量成正比，無緩動。
//   - 捲軸滑塊往下移動，卡片則橫向移動。
// 使用時機: 想在直向的長頁面中，把作品、案例或商品排成一列橫向展示時。
// 標籤: horizontal-scroll, sideways, scrolltrigger, pin, gallery, portfolio, scroll
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#077
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 05
// 05 横スクロール（縦にスクロールすると、カードの列が横へ流れる）
CELLS[4] = (k, w, h, u) => {
  frame(k);
  const y = steps([[0.3, 0], [3.0, 420], [3.5, 420], [4.5, 0]], u);
  const PIN0 = 40, PD = 380, N = 4, PWD = 200, GAP = 20;
  const top = y < PIN0 ? PIN0 - y : 0, p = clamp((y - PIN0) / PD);
  const travel = 34 * 2 + N * PWD + (N - 1) * GAP - VW;
  view(k, () => {
    txt(k, 'WORKS', 34, 30 - y, `800 20px ${EN}`, C.ink, 'left', 'alphabetic', '2px');
    k.save(); k.translate(0, top);
    k.fillStyle = C.soft; k.fillRect(0, 0, VW, VH + 40);
    for (let i = 0; i < N; i++) {
      const x = 34 + i * (PWD + GAP) - p * travel;
      pic(k, x, 18, PWD, 112, i % 3, 8);
      txt(k, '0' + (i + 1), x + 14, 118, `800 22px ${EN}`, '#fff');
      bars(k, x, 140, 120, 1, '#CBD2DE');
    }
    k.restore();
    sbar(k, y / 420, 0.36);
  });
};
