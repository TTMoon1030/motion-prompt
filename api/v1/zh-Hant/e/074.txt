// 捲動淡入 / Fade-in on scroll
// 包: lp (Landing Page 與網站效果) · element 02 · 分類: 捲動時的效果
// 會發生什麼: 頁面捲動時，標題、卡片與內文進入畫面後，各自一邊往上浮一邊淡入。
// 動態規格:
//   - 捲動量：0.35–1.0 秒 0→95 px，1.5–2.1 秒 95→200 px，3.2–4.4 秒 200→0（皆為三次方 easeInOut）。
//   - 觸發線位於視窗高度的 85%（距頂端 143 px）。元素上緣越過這條線時觸發（Intersection Observer 的概念）。
//   - 觸發後 0.7 秒內：不透明度 0→1，位置從下方 28 px → 0（三次方 easeOut）。
//   - 共三個元素：標題（頁面上 172 px 處）、三張卡片並排的一列（206 px）、內文（334 px），一開始都在畫面下方看不到。同一列的卡片同時出現。
//   - 往回捲時，離開畫面下緣的元素會在看不到的地方重設，下次進入畫面時再播放一次。
// 使用時機: 想在長篇 Landing Page 上，隨著讀者捲動逐一呈現內容時。
// 標籤: fade-in, scroll-reveal, reveal-on-scroll, intersection-observer, aos, entrance, scroll
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#074
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 02
// 02 スクロールでフェードイン（画面に入った要素が、下から浮かび上がって現れる）
CELLS[1] = (k, w, h, u) => {
  frame(k);
  const SY = t => steps([[0.35, 0], [1.0, 95], [1.5, 95], [2.1, 200], [3.2, 200], [4.4, 0]], t), y = SY(u);
  const LINE = VH * 0.85, DUR = 0.7, RISE = 28;
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, '暮らしを、軽く。', 34, 52, `800 24px ${JP}`, C.ink); bars(k, 34, 68, 190, 3); btn(k, 34, 108, 96, 26, '詳しく見る');
    pic(k, 286, 18, 196, 110, 0, 8);
    const items = [
      [172, () => txt(k, '選ばれる3つの理由', VW / 2, 172 + 20, `800 18px ${JP}`, C.ink, 'center')],
      [206, () => { for (let i = 0; i < 3; i++) { const x = 34 + i * 152; k.fillStyle = C.soft; rr(k, x, 206, 136, 106, 8); k.fill(); pic(k, x + 8, 214, 120, 56, i, 5); bars(k, x + 10, 280, 112, 2, '#CBD2DE'); } }],
      [334, () => bars(k, 110, 334, 290, 3, C.faint, 7, 13)]
    ];
    for (const [top, draw] of items) {
      const t0 = enteredAt(SY, top, LINE, u); if (t0 === null) continue;
      const q = eOut(P(u, t0, t0 + DUR));
      k.save(); k.globalAlpha = q; k.translate(0, RISE * (1 - q)); draw(); k.restore();
    }
    k.restore();
    sbar(k, y / 200, 0.5);
  });
};
