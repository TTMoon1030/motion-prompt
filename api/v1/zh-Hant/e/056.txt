// Toast / Toast notification
// 包: ui (UI 元件名稱) · element 08 · 分類: 展開與彈出式元件
// 會發生什麼: 點擊「儲存」後，深色通知從右側滑入右下角；下方的剩餘時間線縮完後，通知往右滑出。
// 動態規格:
//   - 4.5 秒循環，卡片 542×191。游標在 1.0 秒點擊左下的藍色「儲存」按鈕（110×36），然後往左下離開。
//   - Toast：#26282c、214×52、圓角 10，位於距右緣 18 px、距底緣 16 px 處；1.05–1.4 秒以 easeOutCubic 從右側滑入 240 px。
//   - 內容：藍色圓形內的白色勾號，以及白色 15 px 文字「已儲存」。
//   - 底部有一條剩餘時間線（高 3 px），1.4–3.2 秒線性地由 194 px 縮短到 0。
//   - 離場：3.2–3.55 秒以 easeInOutCubic 往右滑出 240 px。
// 使用時機: 想在不打斷作業的情況下，簡短告知儲存、送出等結果時使用。
// 標籤: toast, snackbar, notification, alert, message, auto-dismiss
// 尺寸: 542x191 px · 一輪: 4.5 s
// Needs: kit ui -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/ui.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#056
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 08
// 08 トースト（Toast notification）
CELLS[7] = ui((w, h, t) => {
  bar(24, 16, 90, C.line2, 10); strokeRR(24, 34, 300, 34, 7, C.line2, 1.5); bar(24, 82, 90, C.line2, 10); strokeRR(24, 100, 300, 34, 7, C.line2, 1.5);
  fillRR(24, 146, 110, 36, 8, C.blue); text('保存', 79, 164, 16, '#fff', 600, 'center');
  const tin = eo(seg(t, 1.05, 1.4)), tout = eio(seg(t, 3.2, 3.55));
  if (tin > 0 && tout < 1) {
    const tw0 = 214, th = 52, tx = w - 18 - tw0 + (1 - tin) * 240 + tout * 240, ty = h - 16 - th;
    shadowed(() => fillRR(tx, ty, tw0, th, 10, C.dark), 18, 6, .25);
    circle(tx + 26, ty + 24, 11, C.blue); check(tx + 26, ty + 24, 5, '#fff', 2.4);
    text('保存しました', tx + 46, ty + 24, 15, '#fff', 600);
    const life = 1 - seg(t, 1.4, 3.2); fillRR(tx + 10, ty + th - 7, (tw0 - 20) * life, 3, 1.5, C.sb);
  }
  pointer([[.3, -30, h + 30], [.95, 84, 168], [1.5, 84, 168], [2.2, -40, h + 40]], [1.0], t);
});
