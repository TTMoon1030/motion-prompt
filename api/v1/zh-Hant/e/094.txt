// 包: lp (Landing Page 與網站效果) · element 22 · 分類: 載入、轉場與持續動態
// 會發生什麼: 深藍色橫帶中，大型英文服務名稱與作為間隔的橘紅色圓點，不間斷地持續往左流動。
// 動態規格:
//   - 橫帶：高 58，深藍色，與視窗同寬。文字：30 px，粗體，字距 1 px，白色。
//   - 一組為四個詞（WEB DESIGN / BRANDING / MOTION / UI/UX），間隔 64 px，每個間隔中央放一個橘紅色圓點（半徑 6）。
//   - 每個週期（4.8 秒）以等速往左移動一組的寬度（此環境下約 881 px）。實際量測一組的寬度，位移以該寬度循環折返，因此看不到接縫。
//   - 重複排列同一組，直到填滿視窗寬度。
// 使用時機: 想熱鬧地展示服務名稱、客戶 Logo 或標語時。
// 標籤: marquee, ticker, infinite-loop, scrolling-text, logo-strip, band, loop
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#094
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 22
// 22 マーキー（文字の帯が、途切れずに横へ流れ続ける）
CELLS[21] = (k, w, h, u) => {
  frame(k);
  const WORDS = ['WEB DESIGN', 'BRANDING', 'MOTION', 'UI/UX'], F = `800 30px ${EN}`, GAPW = 64;
  view(k, () => {
    bars(k, 34, 20, 180, 1, C.ink, 10); bars(k, 34, 38, 300, 1);
    k.fillStyle = C.ink; k.fillRect(0, 58, VW, 58);
    k.font = F; k.letterSpacing = '1px'; const ws = WORDS.map(s => k.measureText(s).width); k.letterSpacing = '0px';
    const unit = ws.reduce((a, b) => a + b + GAPW, 0);
    let x = -mod(u / CYC, 1) * unit;                 // 1周期で1組ぶん左へ
    for (let rep = 0; x < VW; rep++) WORDS.forEach((s, i) => {
      txt(k, s, x, 88, F, '#fff', 'left', 'middle', '1px');
      k.fillStyle = C.acc; k.beginPath(); k.arc(x + ws[i] + GAPW / 2, 88, 6, 0, 7); k.fill();
      x += ws[i] + GAPW;
    });
    for (let i = 0; i < 3; i++) { k.fillStyle = C.soft; rr(k, 34 + i * 152, 130, 136, 30, 6); k.fill(); bars(k, 46 + i * 152, 142, 90, 1, C.faint); }
  });
};
