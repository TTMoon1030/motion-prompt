// 閱讀進度條 / Scroll progress bar
// 包: lp (Landing Page 與網站效果) · element 07 · 分類: 捲動時的效果
// 會發生什麼: 文章一點一點往下讀時，頂端的線條從左邊延伸，捲到頁面底部時到達另一端。
// 動態規格:
//   - 文章頁捲動 0→430 px，分四段、每段 110 px，每段移動 0.45 秒後停 0.4 秒（三次方 easeInOut）。
//   - 3.75–4.55 秒內由 430→0 捲回原位。
//   - 視窗頂端一條高 6 px 的進度條：底軌為淺灰色，填色為橘紅色。
//   - 填色寬度 = 511 px × 捲動量 / 430，即捲動比例本身（等同 CSS 的 animation-timeline: scroll()）。
// 使用時機: 想在長篇文章中讓讀者知道還剩多少沒讀時。
// 標籤: progress-bar, reading-progress, scroll-progress, indicator, scroll-driven, article
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#079
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 07
// 07 スクロールプログレスバー（読み進めた分だけ、画面上の線が伸びる）
CELLS[6] = (k, w, h, u) => {
  frame(k);
  const MAX = 430;
  const y = steps([[0.3, 0], [0.75, 110], [1.15, 110], [1.6, 220], [2.0, 220], [2.45, 330], [2.85, 330], [3.3, MAX], [3.75, MAX], [4.55, 0]], u);
  view(k, () => {
    k.save(); k.translate(0, -y);
    txt(k, 'はじめてのWebデザイン', 70, 44, `800 20px ${JP}`, C.ink); bars(k, 70, 56, 120, 1, C.faint, 5);
    pic(k, 70, 74, 371, 112, 0, 8);
    for (let b = 0; b < 4; b++) { const y0 = 204 + b * 82; k.fillStyle = C.ink; rr(k, 70, y0, 140, 9, 4.5); k.fill(); bars(k, 70, y0 + 22, 371, 4, C.faint, 6, 12); }
    k.restore();
    k.fillStyle = C.faint; k.fillRect(0, 0, VW, 6);
    k.fillStyle = C.acc; k.fillRect(0, 0, VW * (y / MAX), 6);
    sbar(k, y / MAX, 0.3);
  });
};
