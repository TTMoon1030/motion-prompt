// 漸層動畫 / Animated gradient
// 包: lp (Landing Page 與網站效果) · element 23 · 分類: 載入、轉場與持續動態
// 會發生什麼: 標題後方，橘紅、淺橘紅、藍、淺藍幾團柔和的色塊緩慢繞圈，混色的方式不斷改變。
// 動態規格:
//   - 底色 #FFF1EB，上面疊四個放射狀漸層（中心為該色 → 邊緣透明，各自不透明度 0.85）。
//   - 橘紅色（半徑 230），淺橘紅色 #FFB08F（210），藍色（240），淺藍色 #9DB6FB（200）。
//   - 每個中心點每 4.8 秒沿橢圓繞一圈（x 半徑 70、60、80、50；y 半徑 30、40、30、35；起始角度 0、2.1、4.2、1.0 rad）。
//   - 上方有白色標題（34 px）和按鈕，只有漸層在動。
// 使用時機: 不使用照片，也想讓 Hero 區塊或標題背景有動態與色彩時。
// 標籤: gradient, animated-gradient, mesh-gradient, background, hero, ambient, loop
// 尺寸: 539x208 px · 一輪: 4.8 s
// Needs: kit lp -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/lp.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#095
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 23
// 23 グラデーションアニメーション（背景のグラデーションの色が、ゆっくり流れるように動き続ける）
CELLS[22] = (k, w, h, u) => {
  frame(k);
  const a = 2 * Math.PI * u / CYC;                        // 4.8秒で1周
  const BLOBS = [[C.acc, 150, 60, 230, 70, 30, 0], ['#FFB08F', 380, 40, 210, 60, 40, 2.1], [C.blue, 330, 150, 240, 80, 30, 4.2], ['#9DB6FB', 90, 170, 200, 50, 35, 1.0]];
  view(k, () => {
    k.fillStyle = '#FFF1EB'; k.fillRect(0, 0, VW, VH);
    for (const [col, x0, y0, r, ax, ay, ph] of BLOBS) {
      const x = x0 + ax * Math.cos(a + ph), y = y0 + ay * Math.sin(a + ph);
      const g = k.createRadialGradient(x, y, 0, x, y, r); g.addColorStop(0, col); g.addColorStop(1, 'rgba(255,255,255,0)');
      k.globalAlpha = 0.85; k.fillStyle = g; k.fillRect(0, 0, VW, VH); k.globalAlpha = 1;
    }
    txt(k, 'Hello, Future.', 40, 82, `800 34px ${EN}`, '#fff', 'left', 'alphabetic', '1px');
    bars(k, 40, 100, 200, 2, 'rgba(255,255,255,.8)', 6, 12, 0.6);
    btn(k, 40, 128, 104, 26, 'はじめる', '#fff', C.ink);
  });
};
