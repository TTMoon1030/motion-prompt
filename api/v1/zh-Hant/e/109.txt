// 鏡頭耀光 / Lens flare
// 包: photo (攝影風格：光線、鏡頭與調色) · element 13 · 分類: 鏡頭與相機效果
// 會發生什麼: 太陽從建築後方移到天空中，周圍出現光芒，從太陽穿過畫面中心到另一側，排成一列光點、六角形與光環。
// 動態規格:
//   - 場景 2：街角。後方是一排建築、樹與路燈（x 114, 292, 494），道路在 y 132–162，人物站在左前方（腳底 x 176, y 200，身高 124 px）。
//   - 太陽：半徑 13 px，從建築後方（x 428, y 74）移到開闊的天空（x 474, y 30）
//   - 光芒：8 條，從距太陽中心 17 px 處延伸到 26 px 或 44 px 處，兩種交替（長 9 px 與 27 px；琥珀色，1.6 px，75%）
//   - 光斑：位於從太陽穿過畫面中心的直線上，t = 0.45、0.8、1.15、1.45、1.75、2.0（t = 1 為畫面中心），依序為半徑 7 px 的圓、13 px 的六角形、5 px 的圓、22 px 的六角形、9 px 的圓、30 px 的光環（線寬 3 px）
//   - 顏色為琥珀色 #FFC75A 與 #FFE3A3，不透明度 40–70% × 效果量
//   - 時間：效果在 0.4–1.4 秒進入，維持到 4.0 秒，4.8 秒前恢復原狀（一個循環 5 秒，三次方緩入緩出，約等於 cubic-bezier(0.65,0,0.35,1)）
// 使用時機: 想加入太陽或燈光的眩光、夏日陽光，或電影般的光感時。
// 標籤: lens flare, flare, ghosting, sun flare, starburst, cinematic
// 尺寸: 539x208 px · 一輪: 5 s
// Needs: kit photo -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/photo.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#109
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 13
// レンズフレア: 建物の陰から太陽が出ると、画面の中心をはさんで反対側へ光の玉が並ぶ
CELLS[12] = cell((k, u) => {
  const a = env(u), sx = lerp(428, 474, a), sy = lerp(74, 30, a);
  stageStreet(k, BASE, { sun: [sx, sy, 13, 1] });
  const cx = CW / 2, cy = CH / 2, dx = cx - sx, dy = cy - sy;
  k.save();
  // 太陽の光の筋
  k.globalAlpha = 0.75 * a; k.strokeStyle = BASE.lit; k.lineWidth = 1.6; k.lineCap = 'round';
  for (let i = 0; i < 8; i++) { const t = i / 8 * TAU + 0.2, L = i % 2 ? 26 : 44; k.beginPath(); k.moveTo(sx + Math.cos(t) * 17, sy + Math.sin(t) * 17); k.lineTo(sx + Math.cos(t) * L, sy + Math.sin(t) * L); k.stroke(); }
  // 光の玉: 太陽から中心を通る線の上（t = 0.45〜2.0）
  const G = [[0.45, 7, 'c', 0.6], [0.8, 13, 'h', 0.55], [1.15, 5, 'c', 0.7], [1.45, 22, 'h', 0.4], [1.75, 9, 'c', 0.55], [2.0, 30, 'r', 0.55]];
  for (const [t, r, kind, al] of G) {
    const x = sx + dx * t, y = sy + dy * t;
    k.globalAlpha = al * a;
    if (kind === 'r') { k.strokeStyle = BASE.lit; k.lineWidth = 3; k.beginPath(); k.arc(x, y, r, 0, TAU); k.stroke(); continue; }
    k.fillStyle = kind === 'h' ? BASE.glow : BASE.lit; k.beginPath();
    if (kind === 'h') for (let i = 0; i < 6; i++) { const t2 = i / 6 * TAU; k.lineTo(x + Math.cos(t2) * r, y + Math.sin(t2) * r); } else k.arc(x, y, r, 0, TAU);
    k.closePath(); k.fill();
  }
  k.restore();
});
