// 遮罩轉場 / Mask transition
// 包: motion (Motion Graphics 動態手法) · element 17 · 分類: 轉場與吸睛效果
// 會發生什麼: 圓形或星形的「窗」逐漸擴大，裡面露出下一個場景。
// 動態規格:
//   - 白天場景 → 夜晚場景 → 白天場景，各 2.35 秒。每一半的 0.45–1.6 秒（1.15 秒）進行切換，easeInOut（3 次方）
//   - 在目前場景上方疊上以形狀裁切的下一個場景；形狀擴大到覆蓋整個畫面即完成
//   - 白天→夜晚：圓形，從太陽中心（112, 104）開始，半徑 0→470
//   - 夜晚→白天：5 角星，從月亮中心（427, 104）開始，外半徑 0→1150，擴大的同時旋轉 1.2 rad（約 69°）
//   - 擴大期間，形狀邊緣加上白色外框（線寬 7）
// 使用時機: 想以畫面中的物件（太陽、Logo、按鈕）為起點切換到下一個場景時使用。
// 標籤: mask, transition, iris, wipe, shape-wipe, reveal, scene-change
// 尺寸: 539x208 px · 一輪: 4.7 s
// Needs: kit motion -> https://ttmoon1030.github.io/motion-prompt/api/v1/kit/motion.js
// 會動的範例: https://ttmoon1030.github.io/motion-prompt/#017
//
// cells[i](k, w, h, u)：k = CanvasRenderingContext2D（呼叫端已清除畫面並包好 save/restore），w、h = 尺寸，u = 秒數（0 以上、未滿一輪）。畫面只由 u 決定。
// 以下是元素的程式碼。CELLS、常數與輔助函式都在 kit 裡。

// @element 17
// 17 マスクトランジション（丸や星の形の「窓」が広がり、その中に次の画面が見える）
CELLS[16] = abCell((k, w, h, u, a, b) => {
  const p = eIO(P(u, TA - 0.05, TB + 0.1));
  SCR[a](k, w, h);
  if (p > 0) {
    const c = a === 0 ? SUN : MOON, shape = () => { if (a === 0) { k.beginPath(); k.arc(c[0], c[1], lerp(0, 470, p), 0, 7); } else star(k, c[0], c[1], lerp(0, 1150, p), p * 1.2, 0.5); };
    k.save(); shape(); k.clip(); SCR[b](k, w, h); k.restore();
    if (p < 1) { k.strokeStyle = '#fff'; k.lineWidth = 7; k.lineJoin = 'round'; shape(); k.stroke(); }
  }
});
